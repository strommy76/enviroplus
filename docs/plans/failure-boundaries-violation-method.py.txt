"""One-use real-source counterexamples. Every mutation is restored in finally."""
import json
import os
from pathlib import Path
import subprocess
import sys

E = Path(__file__).resolve().parent
imports = E.parent / 'backup-config-fleet-20261118/test-imports'
S = E/'stocks-worktree'
V = E/'enviroplus-worktree'
records=[]

def observe(name,root,paths,selector,files,mutate=None):
    if len(sys.argv)>1 and name not in sys.argv[1:]:
        return
    saved={p:(root/p).read_bytes() for p in paths}
    try:
        for p in paths:
            if mutate is None:
                revision='c93a27a' if root==S else '0350abe'
                raw=subprocess.check_output(['git','show',revision+':'+p],cwd=root)
            else:
                raw=mutate(p,saved[p].decode()).encode()
            (root/p).write_bytes(raw)
        python=str(S/'.venv/bin/python') if root==S else '/home/pistrommy/.virtualenvs/pimoroni/bin/python'
        arguments=[*files,'-q','-k',selector,'--basetemp='+str(E/(name+'-tmp'))]
        code='import shared; import pytest; raise SystemExit(pytest.main('+repr(arguments)+'))'
        argv=[python,'-c',code] if root==V else [python,'-m','pytest',*arguments]
        with (E/(name+'.log')).open('w') as stream:
            result=subprocess.run(argv,cwd=root,env=os.environ|{'PYTHONPATH':os.pathsep.join([str(imports),str(root)]),'PYTHONDONTWRITEBYTECODE':'1'},stdout=stream,stderr=subprocess.STDOUT)
        record={'name':name,'source_paths':paths,'argv':argv,'exit':result.returncode}
        records.append(record)
        (E/"violation-observations-continuation.json").write_text(json.dumps(records,indent=2)+"\n")
        print(json.dumps(record),flush=True)
        if result.returncode!=1:
            raise RuntimeError(name+' did not produce an assertion/runtime test failure; inspect its log')
    finally:
        for p,raw in saved.items():(root/p).write_bytes(raw)

observe('stocks-lane-violation',S,['src/services/data/freshness.py'],'failed_timestamp_lane or requested_span_problem',['tests/test_data_freshness.py'])

observe('enviro-db-binding-violation',V,['sqlite_backup.py'],'capture_follows_only',['tests/test_sqlite_backup.py'])

observe('stocks-unit-source-violation',S,['src/services/ops/systemd_unit_sources.py'],'job_dependency_and_command',['tests/test_service_execution_units.py'])

(E/'violation-observations.json').write_text(json.dumps(records,indent=2)+'\n')
