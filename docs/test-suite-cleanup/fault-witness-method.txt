rev=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
def run(label,phase,tests):
    cmd=[python,'-m','pytest','-q',*tests,'--basetemp='+str(out/(label+'-'+phase+'-tmp')),'-o','cache_dir='+str(E/'pytest-cache')]
    result=subprocess.run(cmd,cwd=root,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
    (out/(label+'-'+phase+'.log')).write_text(result.stdout)
    return result
with (out/'results.jsonl').open('x') as record:
    for label,file,patches,tests in CASES:
        path=root/file;original=path.read_bytes();altered=original.decode()
        for old,new in patches:
            assert altered.count(old)==1,(label,old)
            altered=altered.replace(old,new)
        baseline=run(label,'before',tests);assert baseline.returncode==0,(label,baseline.stdout)
        try:
            path.write_text(altered)
            observed=run(label,'fault',tests)
        finally:path.write_bytes(original)
        restored=run(label,'restored',tests)
        entry={'revision':rev,'property':label,'source':file,'patches':patches,'tests':tests,'before_rc':baseline.returncode,'fault_rc':observed.returncode,'restored_rc':restored.returncode,'observed_assertion_failure':observed.returncode==1 and 'AssertionError' in observed.stdout and 'FAILED tests/' in observed.stdout and 'ERROR collecting' not in observed.stdout,'source_restored_exact':path.read_bytes()==original}
        record.write(json.dumps(entry)+'\n');record.flush()
        print(label,entry['before_rc'],entry['fault_rc'],entry['restored_rc'],entry['observed_assertion_failure'],flush=True)
        assert entry['observed_assertion_failure'] and entry['restored_rc']==0 and entry['source_restored_exact'],entry
